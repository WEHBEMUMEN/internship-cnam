\section*{Appendix A: Analytical Derivatives of NURBS Basis Functions}
\label{sec:appendix_nurbs_derivatives}

Explicit 1D and 2D analytical derivative expressions used during spatial Jacobian mapping and strain tensor evaluations are stated below.

Analytical spatial derivatives of univariate B-spline basis functions $N_{i,p}(\xi)$ are computed via derivative recurrence:

\begin{equation}
    \frac{d}{d\xi} N_{i,p}(\xi) = p \left( \frac{N_{i,p-1}(\xi)}{\xi_{i+p} - \xi_i} - \frac{N_{i+1,p-1}(\xi)}{\xi_{i+p+1} - \xi_{i+1}} \right)
\end{equation}

For rational NURBS curves with control weights $w_i$, the basis function derivatives are:

\begin{equation}
    \frac{d R_i(\xi)}{d\xi} = w_i \frac{N'_i(\xi) W(\xi) - N_i(\xi) W'(\xi)}{[W(\xi)]^2}, \quad W(\xi) = \sum_{j=0}^{n-1} N_j(\xi) w_j
\end{equation}

For 2D bivariate tensor-product NURBS patches, partial derivatives are:

\begin{align}
    \frac{\partial R_{i,j}(\xi, \eta)}{\partial \xi} &= w_{i,j} \frac{N'_i(\xi) M_j(\eta) W(\xi, \eta) - N_i(\xi) M_j(\eta) \frac{\partial W}{\partial \xi}}{[W(\xi, \eta)]^2} \\
    \frac{\partial R_{i,j}(\xi, \eta)}{\partial \eta} &= w_{i,j} \frac{N_i(\xi) M'_j(\eta) W(\xi, \eta) - N_i(\xi) M_j(\eta) \frac{\partial W}{\partial \eta}}{[W(\xi, \eta)]^2}
\end{align}
